def check(test,label):
    if not test:raise ValueError(label)
def sha(raw):return hashlib.sha256(raw).hexdigest()
def logical(obj):return sha(json.dumps(obj,sort_keys=True,separators=(',',':'),allow_nan=False).encode())
def read(p):return json.loads(Path(p).read_bytes())
def fmt(x):
    if x is None:return '—'
    v=Fraction(str(x))
    if v==int(v):return str(int(v))
    if 0<abs(v)<Fraction('1e-6'):return f'{float(v):.3e}'
    return f'{float(v):.6f}'.rstrip('0').rstrip('.')
def signed(x):
    t=fmt(x)
    return '+'+t if x is not None and Fraction(str(x))>0 else t
def vec(x):return '/'.join(fmt(a) for a in x)
def fullname(k):return k.split('/')[0]+'.'+k.split('/')[-1]
def delta_values(v):return '/'.join(signed(a) for a in v)
def effects(s):
    if not s['all_writes']:return '无'
    return '，'.join(f'{kind}@{fmt(t)} {fmt(before)}→{fmt(after)}' for t,kind,before,after in s['all_writes'])
def stages(s):
    return ' → '.join(f'{phase} {fmt(issue)}/{fmt(end)}/{fmt(ack)}' for phase,issue,end,ack in s['stages']) or '全窗暂缓'
def firsts(s):
    a=[]
    for kind,types in sorted(s['first'].items()):
        a.extend(f'{kind[0]}.{k}@{fmt(t)}' for k,t in sorted(types.items()))
    return '，'.join(a) or '无'
def state_at(s,t):
    return '/'.join(TYPES.get(s['states'][t].get(k),s['states'][t].get(k,'未开题')) for k in ('C/c0','N/n0'))
def close_changes(row):
    if not row.get('lost') and not row.get('gained'):return '无'
    def labels(xs):
        out=[]
        for name,when,typ,*t in xs:
            mark=('首次@'+fmt(t[0]) if when=='first-at' and t else '曾达' if when=='ever' else str(when))
            out.append(f'{name[0]}.{typ}({mark})')
        return ', '.join(out)
    return '；'.join(k+labels(row[x]) for k,x in [('失去 ','lost'),('多得 ','gained')] if row.get(x))

def render_pairs(result):
    text=['# 冷态研究逐配对结果','',
        '所有差值均为比较臂减 J。L 是全窗服务缺额的秒数，U 是未交付服务单位·秒，需求恒为 1。表中列出独立分段率积分，数值最多保留六位小数，微小非零尾差用科学记数法；精确有理数、余额差及分段率差均保存在[结果账](result.json)。资源为真实人员秒／工具秒／发行费。t* 是任一臂首次实际生效，tΔ 是首次不同作用；“整窗”表示相同物理作用下核验至 H12。每项的首次、5／7.5／12 秒当前关闭和完整行动链见[逐臂账](per-arm.md)。','']
    for kind in METHODS:
        pairs=[p for p in result['comparisons'] if p['comparison_type']==kind]
        text+=['## '+kind+f'（{len(pairs)} 对）','',
            '| 条件 | J 的 L/U | 对照 L/U | 服务差 ΔL/ΔU | 判定 | J 人/具/费 | 对照人/具/费 | 资源差 | 首次不同发行（时刻，J／对照） | t*／tΔ，核对至（历元／状态面） | 关闭保有变化 |',
            '|---|---:|---:|---:|---|---:|---:|---:|---|---|---|']
        for p in pairs:
            l,r=p['candidate'],p['comparator'];s=p['service_levels'];resources=p['resource_levels'];official=p['official']
            if s[l] is None or s[r] is None:
                text.append(f"| {p['row']} | — | — | — | {LABEL[p['classification']]} | — | — | — | — | — | — |");continue
            d=official.get('first_issue_difference')
            split=(fmt(d['time'])+'，'+(fullname(d['candidate']) if d['candidate'] else 'W')+'／'+(fullname(d['comparator']) if d['comparator'] else 'W')) if d else '全窗同序' if 'first_issue_difference' in official else '未判'
            times=(fmt(official.get('t_star'))+'／'+fmt(official.get('t_delta'))+'；'+('整窗 12' if official.get('through')==12 else fmt(official.get('through'))))
            if 'points' in official and 'faces' in official:times+=f"（{official['points']}/{official['faces']}）"
            rawdelta=official.get('service_delta',{}).get('rate')
            resource_delta=official.get('resource_delta')
            label=f"[{p['row']}](per-arm.md#{p['row'].lower()})"
            text.append('| '+' | '.join([label,vec(s[l]['rate']),vec(s[r]['rate']),delta_values(rawdelta) if rawdelta else '未判',
                  LABEL[p['classification']],vec(resources[l]),vec(resources[r]),delta_values(resource_delta) if resource_delta else '未判',split,times,close_changes(official) if 'closures_preserved' in official else '未判'])+' |')
        text.append('')
    return '\n'.join(text).rstrip()+'\n'


def render_arms(result):
    out=['# 冷态研究逐臂工作、服务与具名关闭','',
       '阶段 I、A、R 分别为付费初检、具名处置和本人复检。每项依次记录发行／物理终止／首个合法终态回执到达的秒数。状态依次列 C、N 两个工单。NA 表示本人证据支持无需处置，PV 表示本人处置后复核通过，O 表示仍未解决，E 表示已过期。当前 L/U 以独立分段率计算；完整原生余额、每个时点的状态与原始数值见[结果账](result.json)。','']
    for row in result['rows']:
        spec=row['specification'];j=row['job']
        out+=[f"## {row['row']}",'',
            f"公开假设表 {spec['public_law']}，开题 {spec['opening']}，入场上限 {vec(spec['budget'])}。私有评价背景为滤阻 {spec['filter']}、网络 {fmt(spec['network'])}、存储 {fmt(spec['storage'])}、整流输入上限 {spec['rectifier_input_W']} W。"+(f"Slurm 作业 {j['job']}，{j['state']} {j['exit_code']}。" if j else '没有确证作业号。'),'',
            '| 处理 | 全窗 L/U | 人/具/费 | 入场人/具/费 | 实际阶段，发行/终止/回执 | 写前→写后 | 具名首次关闭 | 5 s C/N | 7.5 s C/N | 12 s C/N |',
            '|---|---:|---:|---:|---|---|---|---|---|---|']
        for k,v in row['positions'].items():
            s=v.get('actual')
            if s is None:
                out+=['| '+' | '.join([k.split('-',1)[1],'—','—','—',v['state'],'—','—','—','—','—'])+' |'];continue
            out+=['| '+' | '.join([k.split('-',1)[1],vec(s['service']['rate']),vec(s['resources']),vec(s['admission']),stages(s),effects(s),firsts(s),
                                  state_at(s,'5'),state_at(s,'7.5'),state_at(s,'12')])+' |']
        out+=['']
    return '\n'.join(out).rstrip()+'\n'
