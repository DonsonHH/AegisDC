def blob(name):

def hash_bytes(value):
    return hashlib.sha256(value).hexdigest()
def canon(doc):
    return hash_bytes(json.dumps(doc,sort_keys=True,separators=(',',':'),allow_nan=False).encode())
def path(doc,keys):
    for k in keys:doc=doc[k]
    return doc
def patch(doc,edits):
    for keys,value in edits:path(doc,keys[:-1])[keys[-1]]=copy.deepcopy(value)
def scoped(doc,edits):
    for e in edits:
        keys=e['path']
        if e['op']=='append':
            assert len(path(doc,keys))==e['old_length']
            path(doc,keys).append(copy.deepcopy(e['value']))
        else:
            assert canon(path(doc,keys))==canon(e['was']),keys
            if 'parameter_id' in e:assert path(doc,keys[:-1])['id']==e['parameter_id']
            path(doc,keys[:-1])[keys[-1]]=e['value']

        'action_commands_sha256':canon([]),'duration_s':float(doc['duration_s']),'maximum_step_s':float(doc['step_s'])}
