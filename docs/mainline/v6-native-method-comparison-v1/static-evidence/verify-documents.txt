"""Independent, static Git/JSON readback of a protocol document. No project modules imported."""
from collections import Counter
from copy import deepcopy
import hashlib
import json
from pathlib import Path
import posixpath
import re
import subprocess
import sys

tree,receipt=map(Path,sys.argv[1:3])
pack='docs/mainline/v6-native-method-comparison-v1/'
baseline='ef9ff7e66427d01d5be263e79063897e478e2b54'
old='docs/mainline/v6-intake-engineering-v1/'
close='docs/mainline/v6-intake-expiry-closeout-v1/'
def git(*args):return subprocess.check_output(['git','-C',str(tree),*args])
rev=git('rev-parse','HEAD').decode().strip()
def blob(revision,path):return git('cat-file','blob',revision+':'+path)
def sha(data):return hashlib.sha256(data).hexdigest()
def canonical(value):return sha(json.dumps(value,sort_keys=True,separators=(',',':'),allow_nan=False).encode())
def document(path):return json.loads((tree/path).read_bytes())
def exact(condition,reason):
    if not condition:raise ValueError(reason)
def get(obj,parts):
    for part in parts:obj=obj[part]
    return obj
def loadgit(path):return json.loads(blob(baseline,path))
def amend(obj,edits):
    for path,value in edits:get(obj,path[:-1])[path[-1]]=deepcopy(value)
def change(obj,edits):
    for e in edits:
        where=e['path']; target=get(obj,where)
        if e['op']=='replace':
            exact(canonical(target)==canonical(e['was']),str(where))
            if 'parameter_id' in e:exact(get(obj,where[:-1])['id']==e['parameter_id'],'parameter binding')
            get(obj,where[:-1])[where[-1]]=deepcopy(e['value'])
        elif e['op']=='append':
            exact(len(target)==e['old_length'],'array append identity')
            target.append(deepcopy(e['value']))
        else:raise ValueError(e['op'])
diff=git('diff','--name-only',baseline,rev).decode().splitlines()
allowed={'docs/MAINLINE.md','docs/current-research.json','docs/mainline/progress.json',close+'acceptance-20261120.json'}
exact(all(x in allowed or x.startswith(pack) for x in diff),'files outside document scope')
delivered={}
links=set()
for name in diff:
    raw=(tree/name).read_bytes()
    exact(raw==blob(rev,name),'fresh bytes: '+name)
    delivered[name]={'bytes':len(raw),'git_blob':git('rev-parse',rev+':'+name).decode().strip(),'sha256':sha(raw)}
    if name.startswith(pack) and name.endswith('.md'):
        for link in re.findall(r'\[[^\]]*\]\(([^)]+)\)',raw.decode()):
            target=posixpath.normpath(posixpath.join(posixpath.dirname(name),link.split('#')[0]))
            git('cat-file','-e',rev+':'+target);links.add(target)
c=document(pack+'conditions.json')
for name,digest in c['sources'].items():
    exact(sha(blob(rev,name))==digest and sha((tree/name).read_bytes())==digest,'static input '+name)
history=loadgit(old+'inputs.json')
binding=loadgit(old+'input-binding-addendum.json')
oldcalls={}
for key,item in history['units'].items():
    doc=deepcopy(history['common'] if item['base']=='common' else loadgit(history['legacy_path']))
    amend(doc,item['edits'])
    exact(canonical(doc)==item['sha256'],'original freeze '+key)
    amended=binding['units'][key]
    amend(doc,amended['edits'])
    exact(canonical(doc)==amended['sha256'],'original amendment '+key)
    oldcalls[key]=doc
prior_ids={canonical(v):k for k,v in oldcalls.items()}
prior_groups={(canonical(v['scenario']),canonical(v['boundary_events'])) for v in oldcalls.values()}
all_ids=[];policy_ids=set();phys={};slots=0;ticks=0
for row in c['conditions']:
    task=deepcopy(oldcalls[row['source_unit']]);exact(row['source_call_sha256']==canonical(task),'source unit')
    change(task,c['constructor']['common_edits'])
    change(task,row['row_edits'])
    dd=task['dispatch_program'];pp=dd['policy_program'];biz=task['business_program']
    exact(len(task['scenario']['domains'])==8 and set(task['scenario']['graphs'])=={'physical','functional','control_information','access'},'facility identity')
    exact(task['duration_s']==10 and task['step_s']==0.01 and len(pp['templates'])==1 and pp['templates'][0]['max_instances']==2,'horizon/template')
    actual_slots=len(biz['components']['instance_closure']['episodes']);slots+=actual_slots
    exact(actual_slots==(2 if row['id']=='C17' else 1),'episode denominator')
    clock=dd['decision_times_s'];exact(len(clock)==20 and clock==pp['value_model']['clock_s'],'clock identity')
    ticks+=len(clock)*5
    if row['id']=='C17':
        rec=dd['evidence_ledger']['records'];exact(len(rec)==3 and rec[2]['arrived_at_s']==0.5 and rec[2]['valid_until_s']==0.75 and rec[1]['provenance']['copied_from_record_id']==rec[0]['id'],'limited contention')
    observed={}
    for arm,tok in c['constructor']['arm_tokens'].items():
        val=deepcopy(task);outer=val['dispatch_program'];prog=outer['policy_program']
        prog.update(schema=tok['schema'],policy=tok['policy'])
        if tok['value_model']=='remove_from_policy':del prog['value_model']
        prog['inspection_declaration']=deepcopy(outer['inspection_program']['profile'])
        for y in (prog,outer['closure_program']):y['intervention_declaration']=deepcopy(outer['intervention_program'])
        bus=val['business_program']
        run=bus['run_binding']; ref=bus['components']['diagnostic']['reference']
        reg={'scenario':val['scenario'],'boundary_events':val['boundary_events'],'nominal_knowledge':outer['nominal_knowledge'],
                'ledger':outer['evidence_ledger'],'workforce_profile':prog['workforce_profile'],'inspection_profile':outer['inspection_program']['profile']}
        ref.update({k+'_sha256':canonical(v) for k,v in reg.items()})
        run.update(scenario_sha256=canonical(val['scenario']),boundary_events_sha256=canonical(val['boundary_events']),dispatch_program_sha256=canonical(outer),
                    action_commands_sha256=canonical([]),duration_s=float(val['duration_s']),maximum_step_s=float(val['step_s']))
        ids=dict(complete_call=canonical(val),scenario=canonical(val['scenario']),events=canonical(val['boundary_events']),policy_program=canonical(prog),
                 dispatch=canonical(outer),business=canonical(bus))
        exact(ids==row['arms'][arm],'call identity '+row['id']+'/'+arm)
        observed[arm]=ids
        all_ids.append(ids['complete_call']);policy_ids.add(ids['policy_program'])
    exact(observed['V']==observed['H'],'V/H input equivalence')
    g={(z['scenario'],z['events']) for z in observed.values()};exact(len(g)==1,'within-condition exogenous pair')
    world=g.pop();phys.setdefault(world,[]).append(row['id'])
    exact(not any(z['complete_call'] in prior_ids for z in observed.values()) and world not in prior_groups,'original 19 exact comparison')
summary={'rows':len(c['conditions']),'scheduled_labels':len(all_ids),'nominal_thresholds_and_paths_only':True,'distinct_call_documents':len(set(all_ids)),
         'distinct_policy_documents':len(policy_ids),'physical_groups':len(phys),'public_decision_cap':ticks,'per_arm_instance_opportunities':slots}
exact((summary['rows'],len(all_ids),len(set(all_ids)),len(policy_ids),len(phys),ticks,slots)==(17,85,65,41,5,1700,18),'protocol denominator')
exact(dict(Counter(r['coverage_family'] for r in c['conditions']))==c['selection']['classes'],'predeclared family count')
exact(phys=={(a['scenario_sha256'],a['events_sha256']):a['rows'] for a in c['selection']['shared_physical_groups']},'physical grouping')
accept=document(close+'acceptance-20261120.json')
for name,pin in accept['preserved_sha256'].items():exact(sha(blob(rev,name))==sha(blob(baseline,name))==pin,'preservation '+name)
comp=accept['original_comparison']
exact(comp['accepted_unit_assertions']==[309,309] and comp['original_full_assertions']==[316,325] and comp['unmatched_unit_assertions']==[7,16],'original denominators')
exact({a['id'] for a in accept['historical_jobs_preserved']}=={'134654','134657','134658'},'failed attempts')
slice_paths=['tests/fixtures/39c-business-record-replay-57.json','docs/mainline/39c-business-record-replay-contract-57.md','evidence/mainline/engineering/39c-development/slice-57/result.json']
protected={name:sha(blob(rev,name)) for name in slice_paths}
for name,pin in protected.items():exact(pin==sha(blob(baseline,name)),name)
pg=document('docs/mainline/progress.json'); exact(pg['v6_intake_expiry_closeout']['closed'] and not pg['active_implementations'] and
    pg['v6_native_comparison_v1']['current_authorization']['research_execution'] is False and pg['current_milestone']=='STOP_AT_NATIVE_COMPARISON_PROTOCOL_REVIEW','review gate')
report=dict(schema='aegisdc-v6-native-protocol-static-receipt-v1',content_revision=rev,baseline=baseline,
    scope='Fresh detached Git bytes, Markdown targets, independent deterministic JSON construction and static budget/count arithmetic. No project code or scientific evaluation executed.',
    delivered=delivered,unique_markdown_targets=sorted(links),source_identities_checked=len(c['sources']),prior_complete_calls_checked=len(prior_ids),
    schedule=summary,slice57_sha256=protected,old_source_evidence_protocols_results_and_failures_unchanged=True,
    research_and_qualification_calls_this_package=dict(model=0,facility=0,public_policy=0,engineering_qualification=0,research=0),
    incidental_tools={'editorial_materialization_sha256':sha(Path(sys.argv[3]).read_bytes()),'fresh_verifier_sha256':sha(Path(__file__).read_bytes())},
    stop='PROTOCOL_REVIEW')
with receipt.open('x',encoding='utf-8',newline='\n') as f:
    json.dump(report,f,indent=2,ensure_ascii=False);f.write('\n')
print('Fresh files',len(diff),'linked targets',len(links),'originals',len(prior_ids),'arm documents',len(all_ids),'distinct',len(set(all_ids)),'worlds',len(phys),'slice57',len(protected))
